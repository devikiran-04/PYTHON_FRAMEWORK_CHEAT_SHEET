Select & Filter
basic
Column & row selection
# Single column → Series
df['name']

# Multiple columns → DataFrame
df[['name', 'salary']]

# loc: label-based
df.loc[0, 'name']          # single cell
df.loc[0:3, 'name':'age']  # slice (inclusive!)

# iloc: position-based
df.iloc[0, 1]          # row 0, col 1
df.iloc[1:4, 0:3]      # rows 1-3, cols 0-2
df.iloc[:, -1]         # last column
Key difference
# loc INCLUDES end of slice, iloc EXCLUDES it
Boolean filtering
# Single condition
df[df['age'] > 28]

# Multiple conditions (use & | ~)
df[(df.age > 25) & (df.salary > 60000)]
df[(df.dept == 'IT') | (df.dept == 'Finance')]
df[~(df.dept == 'HR')]  # NOT HR

# isin — match list of values
df[df['dept'].isin(['IT', 'Finance'])]

# between
df[df['age'].between(25, 35)]
Important
# Always use & | ~ not and or not with DataFrames
query() — readable filters
# Clean string-based filtering
df.query('age > 25 and salary > 60000')

# Use @ for Python variables
min_age = 28
df.query('age > @min_age and dept == "IT"')

# Column names with spaces: use backticks
df.query('`first name` == "Alice"')

# Not isin
df.query('dept not in ["HR"]')
Tip
# query() is more readable than chained boolean masks

Transform & Apply
intermediate
apply() — row/col functions
# Apply to a single column
df['salary_k'] = df['salary'].apply(
    lambda x: x / 1000
)

# Apply to each row (axis=1)
df['seniority'] = df.apply(
    lambda row: 'Senior' if row['age'] > 30
                else 'Junior',
    axis=1
)

# Apply a named function
def tax(salary):
    return salary * 0.2 if salary > 60000 else salary * 0.15
df['tax'] = df['salary'].apply(tax)
Output column
0 Junior 1 Senior 2 Senior Name: seniority, dtype: object
map() and replace()
# map: element-wise transformation via dict
df['dept_code'] = df['dept'].map({
    'HR': 1, 'IT': 2, 'Finance': 3
})

# replace: swap specific values
df['dept'].replace('HR', 'Human Resources')
df.replace({'dept': {'HR': 'Human Res',
                    'IT': 'Technology'}})
map vs apply
# map: element-wise on Series # apply: can be row-wise or col-wise
Vectorized with np.where
import numpy as np

# 2-way conditional (fast, vectorized)
df['grade'] = np.where(
    df['score'] >= 60, 'Pass', 'Fail'
)

# Multi-way: np.select
conditions = [
    df.score >= 90,
    df.score >= 75,
    df.score >= 60
]
choices = ['A', 'B', 'C']
df['letter'] = np.select(conditions, choices,
    default='F')

result = df.groupby('dept').agg(
    avg_salary = ('salary', 'mean'),
    max_salary = ('salary', 'max'),
    min_salary = ('salary', 'min'),

transform() — keep shape
# Add group mean as new column
df['dept_avg'] = df.groupby('dept')['salary']\
    .transform('mean')

# Z-score within group
df['salary_zscore'] = df.groupby('dept')['salary']\
    .transform(lambda x:
        (x - x.mean()) / x.std())
Key difference
# transform returns same shape as input # agg returns one row per group

Parse & create dates

df['date'] = pd.to_datetime(df['date_str'])

Resample & rolling
# Resample daily data to monthly
df.set_index('date').resample('ME')['sales'].\
    sum()

# Rolling window statistics
df['roll_7d'] = df['sales'].\
    rolling(window=7, min_periods=1).mean()

# Exponential weighted mean
df['ewm_7d'] = df['sales'].\
    ewm(span=7).mean()

# Percent change day-over-day

Pipe — method chaining
def clean_names(df):
    df.columns = df.columns.str.lower().\
        str.replace(' ','_')
    return df

result = (df
    .pipe(clean_names)
    .dropna(subset=['salary'])
    .query('age > 25')
    .assign(salary_k=lambda x: x.salary/1000)
    .groupby('dept')['salary_k'].mean()
)

Performance: views vs copies
a = np.arange(10)

# VIEW — shares memory (fast, careful!)
b = a[2:7]
b[0] = 99   # also changes a[2]!
a.base is None  # False if view

# COPY — independent array
c = a[2:7].copy()
c[0] = 99   # doesn't affect a

# Check if sharing memory
np.shares_memory(a, b)  # True
Important rule
# Slices are views. Fancy indexing creates copies.
